cmake_minimum_required(VERSION 3.16)
project(uav_ganyu_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(uav_common_msg REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(ganyu_controller src/ganyu_controller.cpp)
target_include_directories(ganyu_controller PRIVATE include)
target_link_libraries(ganyu_controller Eigen3::Eigen)
target_compile_options(ganyu_controller PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(ganyu_controller rclcpp px4_msgs std_msgs rosgraph_msgs uav_common_msg)
install(TARGETS ganyu_controller DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/keyboard.py DESTINATION lib/${PROJECT_NAME} RENAME ganyu_keyboard)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(bspline_check test/bspline_check.cpp)
  target_include_directories(bspline_check PRIVATE include)
  target_link_libraries(bspline_check Eigen3::Eigen)
  add_test(NAME bspline_check COMMAND bspline_check)
endif()
ament_package()
