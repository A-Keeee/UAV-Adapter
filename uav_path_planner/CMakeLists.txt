cmake_minimum_required(VERSION 3.16)
project(uav_path_planner)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/mission.py DESTINATION lib/${PROJECT_NAME} RENAME waypoint_mission)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES ../assets/gazebo_world/walls_demo.sdf DESTINATION share/${PROJECT_NAME}/worlds)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(planner_core test/test_core.py TIMEOUT 90)
endif()
ament_package()
