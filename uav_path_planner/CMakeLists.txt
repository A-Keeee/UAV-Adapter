cmake_minimum_required(VERSION 3.16)
project(uav_path_planner)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(uav_ganyu_control REQUIRED)
find_package(uav_common_msg REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(TinyXML2 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(planner_core src/core.cpp src/lidar.cpp)
target_include_directories(planner_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(planner_core PUBLIC Eigen3::Eigen tinyxml2::tinyxml2)
ament_target_dependencies(planner_core PUBLIC uav_ganyu_control sensor_msgs)
add_executable(waypoint_mission src/mission.cpp)
target_link_libraries(waypoint_mission planner_core nlohmann_json::nlohmann_json yaml-cpp)
ament_target_dependencies(waypoint_mission rclcpp ament_index_cpp uav_common_msg nav_msgs geometry_msgs
  std_msgs rosgraph_msgs visualization_msgs tf2_ros)
add_executable(lidar_avoidance_check src/avoidance_check.cpp)
target_link_libraries(lidar_avoidance_check planner_core nlohmann_json::nlohmann_json)
ament_target_dependencies(lidar_avoidance_check rclcpp std_msgs nav_msgs)
add_executable(unknown_obstacle_scene src/unknown_obstacle_scene.cpp)
target_link_libraries(unknown_obstacle_scene planner_core nlohmann_json::nlohmann_json yaml-cpp)
ament_target_dependencies(unknown_obstacle_scene rclcpp ament_index_cpp std_msgs rosgraph_msgs)
foreach(target planner_core waypoint_mission lidar_avoidance_check unknown_obstacle_scene)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS planner_core waypoint_mission lidar_avoidance_check unknown_obstacle_scene
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES ../assets/gazebo_world/walls_demo.sdf DESTINATION share/${PROJECT_NAME}/worlds)
install(FILES ../assets/gazebo_world/unknown_obstacles.yaml DESTINATION share/${PROJECT_NAME}/scenarios)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(planner_check test/planner_check.cpp TIMEOUT 90)
  target_link_libraries(planner_check planner_core yaml-cpp)
  target_compile_definitions(planner_check PRIVATE TEST_WORLD="${CMAKE_CURRENT_SOURCE_DIR}/../assets/gazebo_world/walls_demo.sdf")
  target_compile_definitions(planner_check PRIVATE TEST_SCENARIO="${CMAKE_CURRENT_SOURCE_DIR}/../assets/gazebo_world/unknown_obstacles.yaml")
endif()
ament_package()
